import fs from 'node:fs';
import path from 'node:path';
import zlib from 'node:zlib';
import crypto from 'node:crypto';
import { execFileSync } from 'node:child_process';

const ROOT = process.cwd();
const IMPORT_ROOT = path.join(ROOT, 'imports-official');

if (!fs.existsSync(IMPORT_ROOT)) {
  console.log('ℹ️ لا يوجد imports-official.');
  process.exit(0);
}

const jobs = fs.readdirSync(IMPORT_ROOT, { withFileTypes: true })
  .filter((entry) => entry.isDirectory())
  .map((entry) => path.join(IMPORT_ROOT, entry.name))
  .filter((dir) => fs.existsSync(path.join(dir, 'READY')));

if (!jobs.length) {
  console.log('ℹ️ لا توجد حزم رسمية جاهزة.');
  process.exit(0);
}

function sha256(buffer) {
  return crypto.createHash('sha256').update(buffer).digest('hex');
}

for (const dir of jobs) {
  const id = path.basename(dir);
  const parts = fs.readdirSync(dir)
    .filter((name) => /^part-\d+\.b64$/.test(name))
    .sort();

  if (!parts.length) throw new Error(`${id}: لا توجد أجزاء`);

  const expectedArchiveHash = fs.readFileSync(path.join(dir, 'archive-sha256.txt'), 'utf8').trim().toLowerCase();
  const expectedTarHash = fs.readFileSync(path.join(dir, 'tar-sha256.txt'), 'utf8').trim().toLowerCase();
  const packedBase64 = parts.map((name) => fs.readFileSync(path.join(dir, name), 'utf8').trim()).join('');
  const packed = Buffer.from(packedBase64, 'base64');
  const actualArchiveHash = sha256(packed);

  if (actualArchiveHash !== expectedArchiveHash) {
    throw new Error(`${id}: SHA-256 للأرشيف غير مطابق\nexpected=${expectedArchiveHash}\nactual=${actualArchiveHash}`);
  }

  const tarBuffer = zlib.brotliDecompressSync(packed);
  const actualTarHash = sha256(tarBuffer);

  if (actualTarHash !== expectedTarHash) {
    throw new Error(`${id}: SHA-256 للـ TAR غير مطابق\nexpected=${expectedTarHash}\nactual=${actualTarHash}`);
  }

  const tempRoot = fs.mkdtempSync(path.join('/tmp', 'teacher-userscripts-official-'));
  const tarPath = path.join(tempRoot, 'official.tar');
  const unpackDir = path.join(tempRoot, 'unpacked');
  fs.mkdirSync(unpackDir, { recursive: true });
  fs.writeFileSync(tarPath, tarBuffer);
  execFileSync('tar', ['-xf', tarPath, '-C', unpackDir], { stdio: 'inherit' });

  const sourceRoot = path.join(unpackDir, 'teacher-userscripts-normalized');
  const sourceScripts = path.join(sourceRoot, 'scripts');
  const auditFile = path.join(sourceRoot, 'IMPORT_AUDIT.json');

  if (!fs.existsSync(sourceScripts) || !fs.existsSync(auditFile)) {
    throw new Error(`${id}: محتوى الأرشيف غير مكتمل`);
  }

  fs.cpSync(sourceScripts, path.join(ROOT, 'scripts'), { recursive: true, force: true });
  fs.mkdirSync(path.join(ROOT, 'docs'), { recursive: true });
  fs.copyFileSync(auditFile, path.join(ROOT, 'docs', 'IMPORT_AUDIT.json'));

  fs.rmSync(dir, { recursive: true, force: true });
  fs.rmSync(path.join(ROOT, 'imports-bulk'), { recursive: true, force: true });
  fs.rmSync(tempRoot, { recursive: true, force: true });

  console.log(`✅ ${id}: تم اعتماد المصدر الرسمي`);
  console.log(`   archive SHA-256: ${actualArchiveHash}`);
  console.log(`   tar SHA-256: ${actualTarHash}`);
}
